% Pipeline overview: training path (top) and inference path (bottom) share one
% canonicalisation step, so the network sees identical inputs in both.
\begin{tikzpicture}[
    font=\sffamily\scriptsize,
    node distance=3.2mm and 3.2mm,
    box/.style={draw=#1!70!black, fill=#1!10, rounded corners=2pt, align=center,
                minimum height=10.5mm, text width=19mm, inner sep=2pt},
    data/.style={draw=black!55, fill=black!4, rounded corners=2pt, align=center,
                 minimum height=10.5mm, text width=17mm, inner sep=2pt},
    result/.style={draw=black!70, fill=white, align=center, minimum height=10.5mm,
                text width=17mm, inner sep=2pt, double, double distance=0.6pt},
    arr/.style={-{Stealth[length=2mm]}, thick, black!65},
    lab/.style={font=\sffamily\tiny\bfseries, text=black!60},
  ]
  % ---------- shared front end (Part A)
  \node[data] (raw) {CT + labels\\ \tiny TotalSeg v3, VerSe19\\ \tiny any orientation};
  \node[box=teal, right=of raw] (load) {\textbf{Load \& audit}\\ \tiny one loader, 2 datasets\\ \tiny 30 anomaly checks};
  \node[box=teal, right=of load] (canon) {\textbf{Canonicalise}\\ \tiny RAS, 2\,mm iso\\ \tiny HU window $\to$ [0,1]};

  % ---------- training path (Part B)
  \node[box=orange, above right=1.5mm and 3.2mm of canon] (crop) {\textbf{Lumbar crop}\\ \tiny from labels\\ \tiny T12..sacrum + 3\,cm};
  \node[box=orange, right=of crop] (train) {\textbf{Train U-Net}\\ \tiny residual encoder\\ \tiny 902 cases, 2.6\,h GPU};
  \node[result,right=of train] (weights) {model\\ weights};

  % ---------- inference path (Parts B, C)
  \node[box=blue, below right=1.5mm and 3.2mm of canon] (seg) {\textbf{Segment}\\ \tiny whole scan, sliding\\ \tiny window, 1 piece/class};
  \node[box=blue, right=of seg] (back) {\textbf{Back to CT grid}\\ \tiny labels L1--L5};
  \node[box=violet, right=of back] (body) {\textbf{Body centre}\\ \tiny erode 6\,mm, keep\\ \tiny body, regrow};
  \node[result,right=of body] (json) {labels.nii.gz\\ centroids.json};

  % ---------- evaluation (Part D)
  \node[box=red, below=4mm of back, text width=48mm] (eval) {\textbf{Evaluate} (original CT grid)\\ \tiny Dice per level $\cdot$ level-shift check $\cdot$ resampling ceiling $\cdot$ centre error};

  % arrows
  \draw[arr] (raw) -- (load);
  \draw[arr] (load) -- (canon);
  \draw[arr] (canon.east) -- ++(1.6mm,0) |- (crop.west);
  \draw[arr] (canon.east) -- ++(1.6mm,0) |- (seg.west);
  \draw[arr] (crop) -- (train);
  \draw[arr] (train) -- (weights);
  \draw[arr, dashed] (weights.south) -- ++(0,-3.2mm) -| (seg.north);
  \draw[arr] (seg) -- (back);
  \draw[arr] (back) -- (body);
  \draw[arr] (body) -- (json);
  \draw[arr] (back.south) -- (eval.north);
  \draw[arr] (load.south) |- (eval.west);
  \draw[arr] (body.south) -- (body.south |- eval.north);

  % lane labels
  \node[lab, above=0.5mm of crop.north west, anchor=south west] {TRAINING (Part B)};
  \node[lab, below=0.5mm of seg.south west, anchor=north west] {INFERENCE (Parts B, C)};
  \node[lab, above=0.5mm of load.north west, anchor=south west] {PREPROCESSING (Part A)};
  \node[lab, below=0.5mm of eval.south west, anchor=north west] {EVALUATION (Part D) \mdseries against the ground-truth labels};
\end{tikzpicture}
